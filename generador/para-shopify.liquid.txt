  function keywords() {
    // se cuenta sin tildes, pero se muestra como está escrito en el aviso
    const raw = S.aviso.toLowerCase().replace(/[^\p{L}\p{N}+#.\s/-]/gu, ' ').split(/\s+/).map(w => w.replace(/^[.\-/]+|[.\-/]+$/g, '')).filter(Boolean);
    const ws = raw.map(norm), show = {};
    const cnt = {};
    const ok = w => w.length >= 3 && !STOP.has(w) && !/^\d+$/.test(w);
    ws.forEach((w, i) => {
      if (!ok(w)) return;
      cnt[w] = (cnt[w] || 0) + 1; show[w] = show[w] || raw[i];
      const n = ws[i + 1]; if (n && ok(n)) { const b = w + ' ' + n; cnt[b] = (cnt[b] || 0) + 1.6; show[b] = show[b] || raw[i] + ' ' + raw[i + 1]; }
    });
    const ranked = Object.entries(cnt).filter(([k, v]) => k.includes(' ') ? v >= 3.2 : true).sort((a, b) => b[1] - a[1]).map(x => x[0]);
    const out = [];
    for (const k of ranked) { if (out.some(o => o.includes(k) || k.includes(o))) continue; out.push(k); if (out.length >= 15) break; }
    return out.map(k => ({ k, label: show[k] }));
  }

    else if (d.verb !== undefined) { const i = +d.verb, cur = S.exp[i].logros; S.exp[i].logros = (cur && !cur.endsWith('\n') ? cur + '\n' : cur) + t.textContent + ' '; save(); render(); const ta = root.querySelector(`textarea[data-i="${i}"]`); if (ta) { ta.focus(); ta.setSelectionRange(ta.value.length, ta.value.length); } }
    else if (d.del) { const [arr, i] = d.del.split('.'); S[arr].splice(+i, 1); save(); render(); }
    else if ('addexp' in d) { S.exp.push({ puesto: '', empresa: '', desde: '', hasta: '', actual: false, logros: '' }); save(); render(); }
    else if ('addedu' in d) { S.edu.push({ titulo: '', inst: '', desde: '', hasta: '', estado: 'Completo' }); save(); render(); }
    else if ('addidi' in d) { S.idiomas.push({ idioma: '', nivel: 'Intermedio' }); save(); render(); }
    else if (d.kw) { S.tecnicas = S.tecnicas.trim() ? S.tecnicas.replace(/[,\s]+$/, '') + ', ' + d.kw : d.kw; save(); $('kwbox').innerHTML = kwHTML(); preview(); }
    else if (d.act === 'perfil') { if (!S.perfil.trim() || confirm('¿Reemplazar tu perfil por un borrador?')) { S.perfil = borradorPerfil(); save(); render(); } }
    else if (d.act === 'pdf') pdf();
    else if (d.act === 'reset') { if (confirm('¿Borrar los datos de esta versión y empezar de cero?')) { S = blank(); save(); go(0); } }
  });
  $('prev').onclick = () => go(step - 1);
  $('next').onclick = () => go(step + 1);
  $('pdf').onclick = pdf;
  window.addEventListener('resize', preview);
  render();
})();
