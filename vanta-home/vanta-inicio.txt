<script>
  (function () {
    var root = document.getElementById('vh-{{ section.id }}');
    if (!root) return;
    /* Packs del producto destacado: se eligen y el botón lleva al producto con ese pack */
    var cta = root.querySelector('.vh-pack-cta');
    function pickPack(label) {
      root.querySelectorAll('.vh-pack').forEach(function (o) { o.classList.remove('is-sel'); });
      label.classList.add('is-sel');
      var r = label.querySelector('input'); if (r) r.checked = true;
      if (cta) {
        cta.href = cta.getAttribute('data-base') + '?pack=' + label.getAttribute('data-pack');
        var t = cta.querySelector('span'); if (t) t.textContent = label.getAttribute('data-cta');
      }
    }
    root.addEventListener('change', function (e) {
      var l = e.target.closest && e.target.closest('.vh-pack'); if (l) pickPack(l);
    });
    root.addEventListener('click', function (e) {
      var l = e.target.closest && e.target.closest('.vh-pack'); if (l) pickPack(l);
    });

    var els = root.querySelectorAll('.vh-reveal');
    if (!('IntersectionObserver' in window) || (window.Shopify && Shopify.designMode)) {
      els.forEach(function (el) { el.classList.add('is-in'); });
      return;
    }
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (e) {
        if (e.isIntersecting) { e.target.classList.add('is-in'); io.unobserve(e.target); }
      });
    }, { rootMargin: '0px 0px -8% 0px' });
    els.forEach(function (el) { io.observe(el); });
    /* Respaldo: si algo no se reveló en 2,5 s, se muestra igual */
    setTimeout(function () { els.forEach(function (el) { el.classList.add('is-in'); }); }, 2500);
  })();
</script>
