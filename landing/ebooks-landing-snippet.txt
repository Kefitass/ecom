window.CE_imgFail = function(img) {
  var box = img.closest("figure") || img.closest(".phone-row");
  if (!box) return;
  var sec = box.closest("section"), list = box.parentNode;
  box.remove();
  if (list && list.classList.contains("peek") && !list.children.length) {
    var head = list.previousElementSibling;

    list.remove();
  }
  if (sec && !sec.querySelector("img")) sec.remove();
};

(function() {
  if (window.CE_BLOCK) return;
  var cfg = window.CE_CONFIG, D = window.CE_DOM;

  function esc(s) { return String(s == null ? "" : s); }

  function topbar() {
    if (document.getElementById("ce-topbar") || !(cfg.mensajesBarra || []).length) return;
    /* Si el tema ya tiene su barra de anuncios (ecom-x / Dawn), no se agrega otra */
    if (!window.CE_PREVIEW && document.querySelector("[id*='announcement-bar'], .announcement-bar, .section-adbar")) return;
    var bar = document.createElement("div");
    bar.id = "ce-topbar";
    var items = cfg.mensajesBarra.map(function(m) { return "<span>" + esc(m) + "</span>"; }).join("");
    bar.innerHTML = '<div class="tk">' + items + items + items + items + "</div>";
    var page = document.getElementById("ce-preview-page");
    var main = page || document.querySelector("main") || document.body.firstElementChild;
    main.parentNode.insertBefore(bar, main);
  }

  function ratingAndBenefits() {
    var title = D.title();
    if (!title || document.getElementById("ce-benefits")) return;
    var after = title.closest(".product__title") || title;
    var r = cfg.rating || {};
    var frag = document.createDocumentFragment();
    if (r.puntuacion > 0 && r.cantidadOpiniones > 0) {
      var rt = document.createElement("div");
      rt.id = "ce-rating";
      rt.innerHTML = '<span class="st" aria-hidden="true">★★★★★</span><b>' + String(r.puntuacion).replace(".", ",") + "</b> · " + r.cantidadOpiniones + " opiniones" +
        (r.lectores > 0 ? " · +" + r.lectores.toLocaleString("es-AR") + " lectores" : "");
      frag.appendChild(rt);
    }
    var ul = document.createElement("ul");
    ul.id = "ce-benefits";
    ul.innerHTML = (cfg.beneficios || []).map(function(b) { return "<li><span>" + b + "</span></li>"; }).join("");
    frag.appendChild(ul);
    after.parentNode.insertBefore(frag, after.nextSibling);
  }

  function setVariant(id) {
    var inp = D.variantInput();
    if (!inp || !window.CE_OK(id)) return;
    inp.value = id;
    inp.dispatchEvent(new Event("change", { bubbles: true }));
  }

  function offers() {
    var btn = D.btn();
    if (!btn || document.getElementById("ce-offers") || !cfg.mostrarOfertas) return;
    var col = btn.closest(".product-form__buttons") || btn.parentNode;
    var w = document.createElement("div");

    w.setAttribute("aria-label", cfg.tituloOfertas || "Opciones");

      w.querySelectorAll(".of").forEach(function(x) { x.classList.remove("sel"); x.setAttribute("aria-checked", "false"); });

      document.dispatchEvent(new CustomEvent("ce:offer", { detail: o }));
    }
    w.querySelectorAll(".of").forEach(function(el) {
      el.addEventListener("click", function() { pick(el); });
      el.addEventListener("keydown", function(e) { if (e.key === " " || e.key === "Enter") { e.preventDefault(); pick(el); } });

  // usando el enlace de carrito de Shopify (/cart/variante:cantidad,variante:cantidad).
  function addon() {
    var a = cfg.agregado || {};
    var off = document.getElementById("ce-offers");
    if (!off || document.getElementById("ce-addon") || !/^\d+$/.test(String(a.varianteId || ""))) return;

      '<span class="pr"><b>' + esc(a.precio) + "</b><s>" + esc(a.precioViejo) + "</s></span>";
    off.parentNode.insertBefore(box, off.nextSibling);
    document.addEventListener("submit", function(e) {
      var chk = document.getElementById("ce-addon-chk");
      if (e.target !== D.form() || !chk || !chk.checked) return;
      if (window.CE_PREVIEW) { e.preventDefault(); e.stopImmediatePropagation(); box.classList.add("ok"); return; }
      var inp = D.variantInput(), main = inp && String(inp.value);
      if (!/^\d+$/.test(main || "")) return;
      e.preventDefault(); e.stopImmediatePropagation();
      window.location.href = "/cart/" + main + ":1," + a.varianteId + ":1";
    }, true);
  }

  function delivery() {
    var off = document.getElementById("ce-offers");
    var btn = D.btn();
    if (!btn || document.getElementById("ce-delivery")) return;
    var col = btn.closest(".product-form__buttons") || btn.parentNode;
    var e = cfg.entrega || {};
    var box = document.createElement("div");
    box.id = "ce-delivery";
    box.innerHTML = '<span class="ic" aria-hidden="true">✉</span><div><b>' + esc(e.titulo) + "</b><span>" + esc(e.texto) +

    var form = D.form();

      box.insertAdjacentElement("afterend", fq);
    }
  }

  function run() {
    if (!D.form()) return false;
    topbar(); ratingAndBenefits(); offers(); addon(); delivery();
    return true;
  }
  var tries = 0;
  var t = setInterval(function() { tries++; if (run() || tries > 40) clearInterval(t); }, 250);
})();

  function sticky() {
    if (!cfg.barraFija || document.getElementById("ce-sticky")) return;
    var bar = document.createElement("div");
    bar.id = "ce-sticky";
    bar.innerHTML = '<div class="in"><div class="tx"><b></b><span></span></div><button type="button">Comprar</button></div>';
    document.body.appendChild(bar);
    function paint(o) { bar.querySelector("b").textContent = o.nombre; bar.querySelector(".tx span").textContent = o.precio; }
    paint(cfg.ofertas.find(function(o) { return o.seleccionado; }) || cfg.ofertas[0]);
    document.addEventListener("ce:offer", function(e) { paint(e.detail); });
    bar.querySelector("button").addEventListener("click", function() {
      var b = D.btn();
      if (b) b.click(); else scrollToBuy();
    });
    function check() {
      var b = D.btn();
      if (!b) return;
      var r = b.getBoundingClientRect();
      bar.classList.toggle("on", r.bottom < 0);
    }
    window.addEventListener("scroll", check, { passive: true });
    check();
  }
