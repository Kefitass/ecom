window.CE_DOM = {
  form: function() {
    return document.querySelector("product-form form[action*='/cart/add']") ||
           document.querySelector("form[id^='product-form-template']") ||
           document.querySelector(".js-product-form") ||
           document.querySelector("form.product-form[action*='/cart/add']") ||

    return (f && f.querySelector(".product-form__submit, button[name='add'], [type='submit']")) ||
           document.querySelector("button[name='add']");
  },
  title: function() {
    return document.querySelector(".product__title h1") || document.querySelector(".product__title") ||
           document.querySelector(".js-product-name") || document.querySelector("h1");
  },
  variantInput: function() {
    var f = this.form();
    return f && f.querySelector("input[name='id'], select[name='id']");
  },
  fullWidth: function() {
    var c = document.getElementById("ce-landing-full");
    if (c) return c;
    var f = this.form();
    if (!f) return null;
    var host = document.getElementById("ce-preview-page") || f.closest(".shopify-section") || f.closest("section");
    c = document.createElement("div");
    c.id = "ce-landing-full";
    if (host && host.parentNode) host.insertAdjacentElement("afterend", c); else f.appendChild(c);
    return c;
  }
};
window.CE_OK = function(v) { return v && String(v).indexOf("⚠️") === -1; };
// Si una imagen de "Mirá por dentro" todavía no está subida a Archivos, se oculta sin dejar huecos.
window.CE_imgFail = function(img) {
  var box = img.closest("figure") || img.closest(".phone-row");
  if (!box) return;
  var sec = box.closest("section"), list = box.parentNode;
  box.remove();
  if (list && list.classList.contains("peek") && !list.children.length) {
    var head = list.previousElementSibling;
    if (head && head.classList.contains("peek-head")) head.remove();
    list.remove();
  }
  if (sec && !sec.querySelector("img")) sec.remove();
};
</script>

  function sticky() {
    if (!cfg.barraFija || document.getElementById("ce-sticky")) return;
    var bar = document.createElement("div");
    bar.id = "ce-sticky";
    bar.innerHTML = '<div class="in"><div class="tx"><b></b><span></span></div><button type="button">Comprar</button></div>';
    document.body.appendChild(bar);
    function paint(o) { bar.querySelector("b").textContent = o.nombre; bar.querySelector(".tx span").textContent = o.precio; }
    paint(cfg.ofertas.find(function(o) { return o.seleccionado; }) || cfg.ofertas[0]);
    document.addEventListener("ce:offer", function(e) { paint(e.detail); });
    bar.querySelector("button").addEventListener("click", function() {
      var b = D.btn();
      if (b) b.click(); else scrollToBuy();
    });
    function check() {
      var b = D.btn();
      if (!b) return;
      var r = b.getBoundingClientRect();
      bar.classList.toggle("on", r.bottom < 0);
    }
    window.addEventListener("scroll", check, { passive: true });
    check();
  }

  var t = setInterval(function() { tries++; if (build() || tries > 40) { clearInterval(t); sticky(); } }, 250);
